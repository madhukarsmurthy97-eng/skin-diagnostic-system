class RecommendationAgent:
    def __init__(self):
        self.advice_dict = {
            "Acne": "Wash gently twice daily. OTC benzoyl peroxide gel may help. Avoid oily cosmetics.",
            "Eczema": "Keep skin moisturized. Mild hydrocortisone cream may relieve itching. Avoid harsh soaps.",
            "Psoriasis": "Apply prescribed ointments. Emollients help reduce scaling. Avoid scratching.",
            "Rosacea": "Use gentle skincare products. Avoid triggers like hot drinks and spicy food.",
            "Fungal Infection": "Keep area dry. Antifungal creams can help. Consult a doctor if persistent.",
            "Vitiligo": "Use sunscreen on affected areas. Gentle skincare is advised.",
            "Hives": "Avoid allergens if known. OTC antihistamines may relieve itching.",
            "Contact Dermatitis": "Avoid irritants. Use mild moisturizers and topical creams as needed.",
            "Shingles": "Seek medical attention. Keep area clean and avoid scratching.",
            "Warts": "Apply OTC salicylic acid treatments. Avoid picking warts.",
            "Cold Sores": "Use antiviral creams. Avoid touching sores to prevent spread.",
            "Hyperpigmentation": "Use sunscreen daily. Gentle exfoliation may help reduce spots.",
            "Dry Skin": "Moisturize frequently. Avoid hot showers and harsh soaps.",
            "Sunburn": "Apply aloe vera or soothing gels. Stay hydrated and avoid further sun exposure.",
            "Multiple/Severe Issues": "Multiple or severe skin issues detected. Please consult a dermatologist immediately."
        }

    def get_advice(self, labels, confs):
        advices, colors = [], []
        for label, conf in zip(labels, confs):
            if label == "Multiple/Severe Issues" or conf < 0.7:
                advices.append("Multiple or severe skin issues detected. Please consult a dermatologist immediately.")
                colors.append("red")
            else:
                advices.append(self.advice_dict.get(label,"Stay safe!"))
                colors.append("green")
        return advices, colors
