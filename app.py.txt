import gradio as gr
from datetime import datetime
from agents.classifier import SkinClassifier
from agents.recommender import RecommendationAgent
from utils.helpers import confidence_bar

class MultiAgentSkinSystem:
    def __init__(self):
        self.classifier = SkinClassifier()
        self.recommender = RecommendationAgent()
        self.history = []
        self.latest_bar = None
    
    def analyze(self, image):
        labels, confs = self.classifier.predict(image)
        advices, colors = self.recommender.get_advice(labels, confs)
        self.latest_bar = confidence_bar(colors, confs, labels)
        report_entries = []
        for l, c, a in zip(labels, confs, advices):
            entry = {
                "date": datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
                "condition": l,
                "confidence": round(c,2),
                "advice": a
            }
            self.history.append(entry)
            report_entries.append(entry)
        return report_entries, self.latest_bar

system = MultiAgentSkinSystem()

def show_report(image):
    if image is None:
        return "No image uploaded", None
    report_entries, bar_img = system.analyze(image)
    report_html = f"""
    <div style='border:2px solid #2E8B57; padding:15px; border-radius:10px; background-color:#F8F8F8'>
        <h2 style='color:#2E8B57; text-align:center;'>🩺 Skin Diagnostic Report</h2>
        <p><b>Date:</b> {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}</p>
        <table style='width:100%; border-collapse: collapse;'>
            <tr>
                <th style='border:1px solid #ccc; padding:5px;'>Condition</th>
                <th style='border:1px solid #ccc; padding:5px;'>Confidence</th>
                <th style='border:1px solid #ccc; padding:5px;'>Advice</th>
            </tr>
    """
    for entry in report_entries:
        color = "red" if entry["confidence"]<0.7 or entry["condition"]=="Multiple/Severe Issues" else "green"
        report_html += f"""
            <tr>
                <td style='border:1px solid #ccc; padding:5px;'>{entry['condition']}</td>
                <td style='border:1px solid #ccc; padding:5px; color:{color};'>{entry['confidence']}</td>
                <td style='border:1px solid #ccc; padding:5px;'>{entry['advice']}</td>
            </tr>
        """
    report_html += "</table></div>"
    return report_html, bar_img

with gr.Blocks() as demo:
    gr.HTML("""
        <div style="background-color:#2E8B57; padding:15px; border-radius:10px">
            <h1 style="color:white; text-align:center;">🩺 Skin Health Diagnostic System</h1>
        </div>
    """)
    
    with gr.Row():
        img_input = gr.Image(type="pil", label="Upload Skin Image")
    report_output = gr.HTML(label="Hospital Report")
    bar_output = gr.Image(label="Confidence Bar / Severity", type="pil")
    
    analyze_btn = gr.Button("🩺 Analyze Skin")
    analyze_btn.click(fn=show_report, inputs=img_input, outputs=[report_output, bar_output])

demo.launch()
